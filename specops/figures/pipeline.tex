% Fig. 1 -- one executable specification, two readings, and the loop that
% turns one marked branch into one test. Wrap in
% \resizebox{\columnwidth}{!}{\input{figures/pipeline}}.
% NB: do not name a style `out' -- pgf reserves /tikz/out for path angles.
\begin{tikzpicture}[font=\footnotesize,>=Stealth,
   srcbox/.style ={draw,rounded corners,align=center,minimum height=6.5mm,
                   inner xsep=4pt,fill=gray!8},
   z3box/.style  ={draw,rounded corners,align=center,minimum height=6.5mm,
                   inner xsep=4pt,fill=yellow!20},
   aibox/.style  ={draw,rounded corners,align=center,minimum height=6.5mm,
                   inner xsep=4pt,fill=orange!16},
   resbox/.style ={draw,rounded corners,align=center,minimum height=6.5mm,
                   inner xsep=4pt,fill=blue!9},
   lbl/.style    ={font=\scriptsize,inner sep=1.5pt}]

  \node[srcbox] (spec) at (0,0)
        {an executable specification in the \rusmt{} DSL,\\[-1pt] with named markers};

  \node[srcbox] (oracle) at (-3.6,-1.3) {\texttt{rustc}\\[-2pt]\emph{concrete oracle}};
  \node[srcbox] (query)  at ( 1.5,-1.3) {transpiler\\[-2pt]marker query $Q_d$};

  \node[z3box] (s1) at (1.5,-2.6)
        {Z3 on $Q_d$, input unconstrained};
  \node[aibox] (s2) at (1.5,-4.0)
        {the model reads $Q_d$, $d$, and the rejected\\[-1pt]
         history, and proposes one candidate $c$};
  \node[z3box] (s3) at (1.5,-5.3)
        {Z3 on $Q_d \wedge (\mathit{input}_0 = c)$};
  \node[resbox] (acc) at (1.5,-6.7)
        {render, replay through the oracle,\\[-1pt] and write the test into the suite};

  \draw[->] (spec.south) -- ++(0,-0.3) -| (oracle.north);
  \draw[->] (spec.south) -- ++(0,-0.3) -| (query.north);
  \draw[->] (query) -- (s1);
  \draw[->] (s1) -- node[lbl,right] {\texttt{unknown} / timeout} (s2);
  \draw[->] (s2) -- (s3);
  \draw[->] (s3) -- node[lbl,right] {\texttt{sat}} (acc);

  \draw[->] (s1.west) -- ++(-2.3,0) node[lbl,above,pos=0.45] {\texttt{sat}}
        |- (acc.west);
  \draw[->] (s1.east) -- ++(1.4,0) node[lbl,above,pos=0.5] {\texttt{unsat}}
        node[lbl,right,pos=1] {$d$ is unreachable};

  \draw[->,dashed] (s3.east) -- ++(1.4,0) node[lbl,above,pos=0.5] {\texttt{unsat}}
        |- (s2.east);

  \draw[->,dashed] (oracle.south) -- ++(0,-5.9)
        node[lbl,left,align=right,pos=0.55] {the same marker\\must fire} -| (acc.south);
\end{tikzpicture}
